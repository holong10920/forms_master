async (page) => {
 const network=[];
 const listener=response=>{const u=new URL(response.url());if(u.pathname.startsWith('/api/') || ['/authorize','/choose'].includes(u.pathname))network.push({method:response.request().method(),path:u.pathname,status:response.status()});};
 page.on('response',listener);
 // Previous stage's synthetic session was expired directly in this module's isolated SQLite DB.
 if(await page.getByRole('link',{name:'Hồ sơ',exact:true}).count()) await page.getByRole('link',{name:'Hồ sơ',exact:true}).click();
 await page.getByRole('link',{name:'Tiếp tục với nhà cung cấp danh tính'}).waitFor();
 const expiredMessage=await page.getByRole('alert').innerText();
 const expiredPath=new URL(page.url()).pathname;
 await page.screenshot({path:'output/playwright/go-auth-session-expired.png'});
 await page.getByRole('link',{name:'Tiếp tục với nhà cung cấp danh tính'}).click();
 await page.getByRole('button',{name:'Continue as synthetic-member'}).click();
 await page.getByRole('button',{name:'Đăng xuất',exact:true}).waitFor();
 await page.getByRole('status').waitFor({state:'hidden'});
 const reloginStatus=await page.evaluate(async()=>(await fetch('/api/v1/auth/me')).status);
 await page.screenshot({path:'output/playwright/go-auth-member-relogin.png'});
 await page.getByRole('button',{name:'Đăng xuất',exact:true}).click();
 await page.getByRole('link',{name:'Tiếp tục với nhà cung cấp danh tính'}).waitFor();
 await page.setViewportSize({width:390,height:844});
 await page.screenshot({path:'output/playwright/go-auth-login-mobile.png',fullPage:true});
 const mobileOverflow=await page.evaluate(()=>document.documentElement.scrollWidth>innerWidth);
 await page.setViewportSize({width:1440,height:1000});
 const finalStatus=await page.evaluate(async()=>(await fetch('/api/v1/auth/me')).status);
 page.off('response',listener);
 if(expiredPath!=='/login'||reloginStatus!==200||finalStatus!==401||mobileOverflow)throw new Error('Expiry checks failed');
 return {scenario:'Server session expiry → login with expiry feedback → re-login → protected workspace → logout',expiredPath,expiredMessage,reloginStatus,finalStatus,mobileOverflow,network};
}
