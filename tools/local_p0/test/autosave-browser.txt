async(page)=>{
 const click=async(p,name)=>{const b=p.getByRole('button',{name,exact:true});await b.focus();await b.press('Enter');};
 const notice=async(p,text)=>p.locator('#notice').filter({hasText:text}).waitFor({timeout:30000});
 const change=async(p,label,value)=>{const el=p.getByLabel(label,{exact:true});await el.fill(value);await el.press('Tab');};
 // Form Studio UI: accounts in the sign-in card; save, reload and autosave in the save menu; forms open from home cards; field types in the "追加" palette; data key in the inspector "詳細" tab.
 const press=async l=>{await l.scrollIntoViewIfNeeded();await l.focus();await l.press('Enter');};
 const setSwitch=async(l,on)=>{if(await l.isChecked()!==on){await l.focus();await l.press('Space');}if(await l.isChecked()!==on)throw Error('switch state');};
 const menu=async(p,id)=>{await p.locator('#save-state').click();await press(p.locator(id));};
 const addField=async(p,type)=>{await p.locator('#left-tabs [data-tab="add"]').click();await press(p.locator(`#palette [data-add-type="${type}"]`));};
 const tab=async(p,key)=>{await p.locator(`#inspector-tabs [data-tab="${key}"]`).click();await p.locator(`#inspector-tabs [data-tab="${key}"][aria-selected="true"]`).waitFor();};
 await page.goto('http://localhost:4174/');await press(page.locator('#account-list [data-account="synthetic-editor"]'));await notice(page,'ログイン完了');await page.locator('#site').selectOption('site_sports_synthetic');await press(page.locator('#new-form'));await page.locator('#new-name').fill('合成自動保存');await click(page,'フォーム作成');await page.locator('#version').filter({hasText:'v1'}).waitFor();const form=new URL(page.url()).searchParams.get('form');if(!await page.locator('#autosave').isChecked())throw Error('autosave default disabled');
 const peer=await page.context().newPage();await peer.goto('http://localhost:4174/');await peer.locator('#studio').waitFor();await peer.locator('#site').selectOption('site_sports_synthetic');await press(peer.locator(`#forms [data-form-id="${form}"]`));await notice(peer,'保存済みドラフトを読み込みました');await peer.locator('#save-state').click();await setSwitch(peer.locator('#autosave'),false);await peer.keyboard.press('Escape');
 const saves=[];page.on('request',r=>{if(r.url().endsWith('/api/save'))saves.push(r.postDataJSON());});
 await addField(page,'text');await change(page,'表示名','合成サーバ項目');await tab(page,'advanced');await change(page,'データキー','syntheticServer');await page.locator('#version').filter({hasText:'v2'}).waitFor();await notice(page,'保存しました');if(saves.length!==1)throw Error('debounce did not coalesce');
 await addField(peer,'text');await change(peer,'表示名','合成ローカル項目');await menu(peer,'#save');await peer.locator('#conflict').waitFor();const details=JSON.parse(await peer.locator('#conflict-details').textContent());if(details.currentVersion!==2||!details.serverChanges.some(s=>s.includes('合成サーバ項目'))||!details.localChanges.some(s=>s.includes('合成ローカル項目')))throw Error('conflict comparison absent');if(await peer.getByLabel('表示名',{exact:true}).inputValue()!=='合成ローカル項目')throw Error('conflict lost local edit');await peer.screenshot({path:'output/playwright/local-p0-cas-conflict.png',fullPage:true});
 await menu(page,'#load');await notice(page,'保存済みドラフトを読み込みました');if(!await page.locator('#canvas').textContent().then(s=>s.includes('合成サーバ項目')&&!s.includes('合成ローカル項目')))throw Error('conflict overwrote server');await peer.close();return{allPassed:true,form,autosaveDefault:true,debouncedRequests:saves.length,casConflict:true,localEditsPreserved:true,serverNotOverwritten:true,title:await page.locator('#form-title').inputValue()};
}
