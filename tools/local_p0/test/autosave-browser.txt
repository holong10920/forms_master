async(page)=>{
 const click=async(p,name)=>{const b=p.getByRole('button',{name,exact:true});await b.focus();await b.press('Enter');};
 const notice=async(p,text)=>p.locator('#notice').filter({hasText:text}).waitFor({timeout:30000});
 const change=async(p,label,value)=>{const el=p.getByLabel(label,{exact:true});await el.fill(value);await el.press('Tab');};
 await page.goto('http://localhost:4174/');await page.locator('#account').selectOption('synthetic-editor');await click(page,'ログイン / 切替');await notice(page,'ログイン完了');await page.locator('#site').selectOption('site_sports_synthetic');await page.locator('#new-name').fill('合成自動保存');await click(page,'フォーム作成');await page.locator('#version').filter({hasText:'v1'}).waitFor();const form=await page.locator('#forms').inputValue();if(!await page.locator('#autosave').isChecked())throw Error('autosave default disabled');
 const peer=await page.context().newPage();await peer.goto('http://localhost:4174/');await peer.locator('#studio').waitFor();await peer.locator('#site').selectOption('site_sports_synthetic');await peer.locator('#forms').selectOption(form);await click(peer,'開く / 再読込');await notice(peer,'保存済みドラフトを読み込みました');await peer.locator('#autosave').uncheck();
 const saves=[];page.on('request',r=>{if(r.url().endsWith('/api/save'))saves.push(r.postDataJSON());});
 await page.locator('#palette').selectOption('text');await click(page,'フィールド追加');await change(page,'表示名','合成サーバ項目');await change(page,'データキー','syntheticServer');await page.locator('#version').filter({hasText:'v2'}).waitFor();await notice(page,'保存しました');if(saves.length!==1)throw Error('debounce did not coalesce');
 await peer.locator('#palette').selectOption('text');await click(peer,'フィールド追加');await change(peer,'表示名','合成ローカル項目');await click(peer,'保存');await peer.locator('#conflict').waitFor();const details=JSON.parse(await peer.locator('#conflict-details').textContent());if(details.currentVersion!==2||!details.serverChanges.some(s=>s.includes('合成サーバ項目'))||!details.localChanges.some(s=>s.includes('合成ローカル項目')))throw Error('conflict comparison absent');if(await peer.getByLabel('表示名',{exact:true}).inputValue()!=='合成ローカル項目')throw Error('conflict lost local edit');await peer.screenshot({path:'output/playwright/local-p0-cas-conflict.png',fullPage:true});
 await click(page,'開く / 再読込');await notice(page,'保存済みドラフトを読み込みました');if(!await page.locator('#canvas').textContent().then(s=>s.includes('合成サーバ項目')&&!s.includes('合成ローカル項目')))throw Error('conflict overwrote server');await peer.close();return{allPassed:true,form,autosaveDefault:true,debouncedRequests:saves.length,casConflict:true,localEditsPreserved:true,serverNotOverwritten:true,title:await page.locator('#form-title').inputValue()};
}
