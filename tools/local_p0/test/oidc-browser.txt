async(page)=>{
 const callbackRequests=[];const clientRequests=[];

 const click=async name=>{const b=page.getByRole('button',{name,exact:true});await b.focus();await b.press('Enter');};

 const checks=await page.evaluate(async()=>{const session=await(await fetch('/api/session')).json();const call=async(op,body)=>{const r=await fetch('/api/'+op,{method:'POST',headers:{'Content-Type':'application/json','X-CSRF-Token':session.csrf},body:JSON.stringify(body)});return{status:r.status,value:await r.json()};};const site='site_sports_synthetic';const forms=await call('list',{site});const form=forms.value.forms.find(f=>f.formId==='frm_sports').formId;const data=await call('get',{site,form});return{user:session.user,read:data.status,write:(await call('save',{site,form,expected:data.value.version,definition:data.value.definition})).status,crossSite:(await call('list',{site:'site_other_synthetic'})).status,storageContainsTokens:[...Object.values(localStorage),...Object.values(sessionStorage)].some(v=>/eyJ[A-Za-z0-9_-]+\./.test(v))};});if(checks.read!==200||checks.write!==403||checks.crossSite!==403||checks.storageContainsTokens)throw Error('viewer RBAC isolation');
 const cookies=await page.context().cookies('http://localhost:4174/');const sessionCookie=cookies.find(c=>c.name==='__Host-session');if(!sessionCookie?.secure||!sessionCookie.httpOnly||sessionCookie.sameSite!=='Strict')throw Error('BFF cookie flags');const replay=await page.goto(callbackRequests.at(-1));if(replay.status()!==401)throw Error('authorization callback replay');await page.goto('http://localhost:4174/');await page.locator('#identity').filter({hasText:'synthetic-viewer'}).waitFor();await click('ログアウト');await page.locator('#studio').waitFor({state:'hidden'});const loggedOut=await page.evaluate(async()=>({status:(await fetch('/api/session')).status}));if(loggedOut.status!==401)throw Error('logout retained session');await login('synthetic-editor');await page.screenshot({path:'output/playwright/closure-oidc-admin.png'});return{allPassed:true,provider:'local mock; not Cognito',authorizationCode:true,pkceS256:true,noBrowserTokenExchange:true,callbackReplayRejected:true,secureHttpOnlyStrictCookie:true,viewerRead:true,viewerWriteDenied:true,crossSiteDenied:true,logoutUnauthorized:true};
